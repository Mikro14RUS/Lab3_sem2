// Array — динамический массив строк. В Go срез уже является
// динамическим массивом, поэтому структура хранит его внутри
// и предоставляет предметный интерфейс ЛР1.
type Array struct {
        name string
        data []string
}

func NewArray(name string) *Array {
        return &Array{name: name, data: make([]string, 0)}
}

func (a *Array) Type() string { return TypeArray }
func (a *Array) Name() string { return a.name }
func (a *Array) Len() int     { return len(a.data) }
func (a *Array) Clear()       { a.data = a.data[:0] }

func (a *Array) PushBack(value string) { a.data = append(a.data, value) }

func (a *Array) PushFront(value string) {
        a.data = append(a.data, "")
        copy(a.data[1:], a.data[:len(a.data)-1])
        a.data[0] = value
}

func (a *Array) InsertAt(index int, value string) bool {
        if index < 0 || index > len(a.data) {
                return false
        }
        a.data = append(a.data, "")
        copy(a.data[index+1:], a.data[index:])
        a.data[index] = value
        return true
}

func (a *Array) PopBack() (string, bool) {
        if len(a.data) == 0 {
                return "", false
        }
        value := a.data[len(a.data)-1]
        a.data = a.data[:len(a.data)-1]
        return value, true
}

func (a *Array) PopFront() (string, bool) {
        if len(a.data) == 0 {
                return "", false
        }
        value := a.data[0]
        a.data = a.data[1:]
        return value, true
}

func (a *Array) RemoveAt(index int) (string, bool) {
        if index < 0 || index >= len(a.data) {
                return "", false
        }
        value := a.data[index]
        copy(a.data[index:], a.data[index+1:])
        a.data = a.data[:len(a.data)-1]
        return value, true
}

func (a *Array) GetAt(index int) (string, bool) {
        if index < 0 || index >= len(a.data) {
                return "", false
        }
        return a.data[index], true
}

func (a *Array) SetAt(index int, value string) bool {
        if index < 0 || index >= len(a.data) {
                return false
        }
        a.data[index] = value
        return true
}

func (a *Array) Values() []string {
        result := make([]string, len(a.data))
        copy(result, a.data)
        return result
}

func (a *Array) String() string {
        if len(a.data) == 0 {
                return "[]"
        }
        return "[" + strings.Join(a.data, ", ") + "]"
}

func (a *Array) MarshalJSON() ([]byte, error) {
        if a.data == nil {
                return []byte("[]"), nil
        }
        return json.Marshal(a.data)
}

func (a *Array) UnmarshalJSON(data []byte) error {
        var values []string
        if err := json.Unmarshal(data, &values); err != nil {
                return fmt.Errorf("array: %w", err)
        }
        a.data = values
        if a.data == nil {
                a.data = make([]string, 0)
        }
        return nil
}

type doublyNode struct {
        value string
        prev  *doublyNode
        next  *doublyNode
}

// DoublyLinkedList — двусвязный список. Благодаря prev удаление
// хвоста и обратный обход выполняются за O(1).
type DoublyLinkedList struct {
        name string
        head *doublyNode
        tail *doublyNode
        size int
}

func NewDoublyLinkedList(name string) *DoublyLinkedList {
        return &DoublyLinkedList{name: name}
}

func (l *DoublyLinkedList) Type() string { return TypeDList }
func (l *DoublyLinkedList) Name() string { return l.name }
func (l *DoublyLinkedList) Len() int     { return l.size }

func (l *DoublyLinkedList) Clear() {
        l.head = nil
        l.tail = nil
        l.size = 0
}

func (l *DoublyLinkedList) PushFront(value string) {
        node := &doublyNode{value: value, next: l.head}
        if l.head != nil {
                l.head.prev = node
        } else {
                l.tail = node
        }
        l.head = node
        l.size++
}

func (l *DoublyLinkedList) PushBack(value string) {
        node := &doublyNode{value: value, prev: l.tail}
        if l.tail != nil {
                l.tail.next = node
        } else {
                l.head = node
        }
        l.tail = node
        l.size++
}

func (l *DoublyLinkedList) InsertBefore(target, value string) bool {
        node := l.findNode(target)
        if node == nil {
                return false
        }
        if node == l.head {
                l.PushFront(value)
                return true
        }
        inserted := &doublyNode{value: value, prev: node.prev, next: node}
        node.prev.next = inserted
        node.prev = inserted
        l.size++
        return true
}

func (l *DoublyLinkedList) InsertAfter(target, value string) bool {
        node := l.findNode(target)
        if node == nil {
                return false
        }
        if node == l.tail {
                l.PushBack(value)
                return true
        }
        inserted := &doublyNode{value: value, prev: node, next: node.next}
        node.next.prev = inserted
        node.next = inserted
        l.size++
        return true
}

func (l *DoublyLinkedList) PopFront() (string, bool) {
        if l.head == nil {
                return "", false
        }
        node := l.head
        l.head = node.next
        if l.head != nil {
                l.head.prev = nil
        } else {
                l.tail = nil
        }
        l.size--
        return node.value, true
}

func (l *DoublyLinkedList) PopBack() (string, bool) {
        if l.tail == nil {
                return "", false
        }
        node := l.tail
        l.tail = node.prev
        if l.tail != nil {
                l.tail.next = nil
        } else {
                l.head = nil
        }
        l.size--
        return node.value, true
}

func (l *DoublyLinkedList) DeleteBefore(target string) (string, bool) {
        node := l.findNode(target)
        if node == nil || node.prev == nil {
                return "", false
        }
        return l.unlink(node.prev), true
}

func (l *DoublyLinkedList) DeleteAfter(target string) (string, bool) {
        node := l.findNode(target)
        if node == nil || node.next == nil {
                return "", false
        }
        return l.unlink(node.next), true
}

func (l *DoublyLinkedList) DeleteValue(value string) (string, bool) {
        node := l.findNode(value)
        if node == nil {
                return "", false
        }
        return l.unlink(node), true
}

func (l *DoublyLinkedList) Find(value string) int {
        index := 0
        for node := l.head; node != nil; node = node.next {
                if node.value == value {
                        return index
                }
                index++
        }
        return -1
}

func (l *DoublyLinkedList) GetAt(index int) (string, bool) {
        node := l.nodeAt(index)
        if node == nil {
                return "", false
        }
        return node.value, true
}

func (l *DoublyLinkedList) SetAt(index int, value string) bool {
        node := l.nodeAt(index)
        if node == nil {
                return false
        }
        node.value = value
        return true
}

func (l *DoublyLinkedList) Values() []string {
        result := make([]string, 0, l.size)
        for node := l.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

func (l *DoublyLinkedList) ValuesReverse() []string {
        result := make([]string, 0, l.size)
        for node := l.tail; node != nil; node = node.prev {
                result = append(result, node.value)
        }
        return result
}

func (l *DoublyLinkedList) String() string {
        if l.size == 0 {
                return "[]"
        }
        return "[" + strings.Join(l.Values(), " <-> ") + "]"
}

func (l *DoublyLinkedList) nodeAt(index int) *doublyNode {
        if index < 0 || index >= l.size {
                return nil
        }
        node := l.head
        for i := 0; i < index; i++ {
                node = node.next
        }
        return node
}

func (l *DoublyLinkedList) findNode(value string) *doublyNode {
        for node := l.head; node != nil; node = node.next {
                if node.value == value {
                        return node
                }
        }
        return nil
}

func (l *DoublyLinkedList) unlink(node *doublyNode) string {
        if node.prev != nil {
                node.prev.next = node.next
        } else {
                l.head = node.next
        }
        if node.next != nil {
                node.next.prev = node.prev
        } else {
                l.tail = node.prev
        }
        l.size--
        return node.value
}

func (l *DoublyLinkedList) MarshalJSON() ([]byte, error) {
        return json.Marshal(l.Values())
}

func (l *DoublyLinkedList) UnmarshalJSON(data []byte) error {
        var values []string
        if err := json.Unmarshal(data, &values); err != nil {
                return fmt.Errorf("dlist: %w", err)
        }
        l.Clear()
        for _, value := range values {
                l.PushBack(value)
        }
        return nil
}

// FullBinaryTree — бинарное дерево поиска с проверкой свойства
// Full Binary Tree (вариант 4 ЛР1): у каждого узла 0 или 2 ребёнка.
type FullBinaryTree struct {
        name string
        root *treeNode
        size int
}

func NewFullBinaryTree(name string) *FullBinaryTree {
        return &FullBinaryTree{name: name}
}

func (t *FullBinaryTree) Type() string { return TypeTree }
func (t *FullBinaryTree) Name() string { return t.name }
func (t *FullBinaryTree) Len() int     { return t.size }

func (t *FullBinaryTree) Clear() {
        t.root = nil
        t.size = 0
}

func (t *FullBinaryTree) Insert(value int) bool {
        inserted := false
        t.root = insertTreeNode(t.root, value, &inserted)
        if inserted {
                t.size++
        }
        return inserted
}

func (t *FullBinaryTree) Find(value int) bool {
        node := t.root
        for node != nil {
                if value == node.value {
                        return true
                }
                if value < node.value {
                        node = node.left
                } else {
                        node = node.right
                }
        }
        return false
}

func (t *FullBinaryTree) IsFull() bool { return isFullNode(t.root) }

func (t *FullBinaryTree) IsBST() bool {
        return isBSTNode(t.root, math.MinInt64, math.MaxInt64)
}

func (t *FullBinaryTree) BFS() []int {
        result := make([]int, 0, t.size)
        if t.root == nil {
                return result
        }
        queue := []*treeNode{t.root}
        for len(queue) > 0 {
                node := queue[0]
                queue = queue[1:]
                result = append(result, node.value)
                if node.left != nil {
                        queue = append(queue, node.left)
                }
                if node.right != nil {
                        queue = append(queue, node.right)
                }
        }
        return result
}

func (t *FullBinaryTree) Preorder() []int {
        result := make([]int, 0, t.size)
        preorderNode(t.root, &result)
        return result
}

func (t *FullBinaryTree) Inorder() []int {
        result := make([]int, 0, t.size)
        inorderNode(t.root, &result)
        return result
}

func (t *FullBinaryTree) Postorder() []int {
        result := make([]int, 0, t.size)
        postorderNode(t.root, &result)
        return result
}

// PreorderWithNulls возвращает preorder-обход с nil-маркерами
// отсутствующих детей. Формат удобен для точного восстановления формы.
func (t *FullBinaryTree) PreorderWithNulls() []*int {
        result := make([]*int, 0, t.size*2+1)
        collectPreorder(t.root, &result)
        return result
}

func (t *FullBinaryTree) LoadPreorderWithNulls(tokens []*int) bool {
        position := 0
        root := buildFromPreorder(tokens, &position)
        if root == nil && position != len(tokens) {
                return false
        }
        if position != len(tokens) {
                return false
        }
        t.root = root
        t.size = countNodes(root)
        return true
}

func (t *FullBinaryTree) String() string {
        return "BFS: [" + joinInts(t.BFS()) + "]"
}

func (t *FullBinaryTree) StringTraversal(mode string) string {
        switch strings.ToUpper(mode) {
        case "PRE":
                return "PRE: " + joinInts(t.Preorder())
        case "IN":
                return "IN: " + joinInts(t.Inorder())
        case "POST":
                return "POST: " + joinInts(t.Postorder())
        case "PRETTY":
                return t.PrettyString()
        default:
                return t.String()
        }
}

func (t *FullBinaryTree) PrettyString() string {
        if t.root == nil {
                return "Дерево пусто"
        }
        var out strings.Builder
        out.WriteString("Дерево (размер: " + strconv.Itoa(t.size) + "):\n")
        prettyNode(t.root, "", true, &out)
        return out.String()
}

func (t *FullBinaryTree) MarshalJSON() ([]byte, error) {
        return json.Marshal(t.PreorderWithNulls())
}

func (t *FullBinaryTree) UnmarshalJSON(data []byte) error {
        var tokens []*int
        if err := json.Unmarshal(data, &tokens); err != nil {
                return fmt.Errorf("tree: %w", err)
        }
        if !t.LoadPreorderWithNulls(tokens) {
                return fmt.Errorf("tree: некорректный preorder-формат")
        }
        return nil
}

func insertTreeNode(node *treeNode, value int, inserted *bool) *treeNode {
        if node == nil {
                *inserted = true
                return &treeNode{value: value}
        }
        if value < node.value {
                node.left = insertTreeNode(node.left, value, inserted)
        } else if node.value < value {
                node.right = insertTreeNode(node.right, value, inserted)
        }
        return node
}

func isFullNode(node *treeNode) bool {
        if node == nil {
                return true
        }
        if (node.left == nil) != (node.right == nil) {
                return false
        }
        return isFullNode(node.left) && isFullNode(node.right)
}

func isBSTNode(node *treeNode, minValue, maxValue int64) bool {
        if node == nil {
                return true
        }
        value := int64(node.value)
        if value <= minValue || value >= maxValue {
                return false
        }
        return isBSTNode(node.left, minValue, value) &&
                isBSTNode(node.right, value, maxValue)
}

func preorderNode(node *treeNode, out *[]int) {
        if node == nil {
                return
        }
        *out = append(*out, node.value)
        preorderNode(node.left, out)
        preorderNode(node.right, out)
}

func inorderNode(node *treeNode, out *[]int) {
        if node == nil {
                return
        }
        inorderNode(node.left, out)
        *out = append(*out, node.value)
        inorderNode(node.right, out)
}

func postorderNode(node *treeNode, out *[]int) {
        if node == nil {
                return
        }
        postorderNode(node.left, out)
        postorderNode(node.right, out)
        *out = append(*out, node.value)
}

func collectPreorder(node *treeNode, out *[]*int) {
        if node == nil {
                *out = append(*out, nil)
                return
        }
        value := node.value
        *out = append(*out, &value)
        collectPreorder(node.left, out)
        collectPreorder(node.right, out)
}

func buildFromPreorder(tokens []*int, position *int) *treeNode {
        if *position >= len(tokens) {
                *position = len(tokens) + 1
                return nil
        }
        token := tokens[*position]
        *position = *position + 1
        if token == nil {
                return nil
        }
        node := &treeNode{value: *token}
        node.left = buildFromPreorder(tokens, position)
        node.right = buildFromPreorder(tokens, position)
        return node
}

func countNodes(node *treeNode) int {
        if node == nil {
                return 0
        }
        return 1 + countNodes(node.left) + countNodes(node.right)
}

func prettyNode(node *treeNode, prefix string, isLast bool, out *strings.Builder) {
        if node == nil {
                return
        }
        connector := "├── "
        if isLast {
                connector = "└── "
        }
        out.WriteString(prefix + connector + strconv.Itoa(node.value) + "\n")
        childPrefix := prefix
        if isLast {
                childPrefix += "    "
        } else {
                childPrefix += "│   "
        }
        if node.left != nil || node.right != nil {
                prettyNode(node.left, childPrefix, node.right == nil, out)
                prettyNode(node.right, childPrefix, true, out)
        }
}

func joinInts(values []int) string {
        parts := make([]string, len(values))
        for i, value := range values {
                parts[i] = strconv.Itoa(value)
        }
        return strings.Join(parts, ", ")
}

type hashEntry struct {
        Key   string `json:"key"`
        Value string `json:"value"`
}

// HashTable — хеш-таблица с разрешением коллизий методом цепочек.
// Используется FNV-1a и автоматическое расширение при load factor > 0.75.
type HashTable struct {
        name       string
        buckets    [][]hashEntry
        size       int
        collisions int
}

func NewHashTable(name string, initialBuckets int) *HashTable {
        if initialBuckets < 2 {
                initialBuckets = 2
        }
        return &HashTable{name: name, buckets: make([][]hashEntry, initialBuckets)}
}

func (h *HashTable) Type() string { return TypeHash }
func (h *HashTable) Name() string { return h.name }
func (h *HashTable) Len() int     { return h.size }

func (h *HashTable) Clear() {
        for i := range h.buckets {
                h.buckets[i] = nil
        }
        h.size = 0
        h.collisions = 0
}

func (h *HashTable) Put(key, value string) bool {
        index := h.bucketIndex(key)
        for i := range h.buckets[index] {
                if h.buckets[index][i].Key == key {
                        h.buckets[index][i].Value = value
                        return false
                }
        }
        if len(h.buckets[index]) > 0 {
                h.collisions++
        }
        h.buckets[index] = append(h.buckets[index], hashEntry{Key: key, Value: value})
        h.size++
        if h.size*4 > len(h.buckets)*3 {
                h.rehash(len(h.buckets) * 2)
        }
        return true
}

func (h *HashTable) Get(key string) (string, bool) {
        for _, entry := range h.buckets[h.bucketIndex(key)] {
                if entry.Key == key {
                        return entry.Value, true
                }
        }
        return "", false
}

func (h *HashTable) Remove(key string) (string, bool) {
        index := h.bucketIndex(key)
        bucket := h.buckets[index]
        for i, entry := range bucket {
                if entry.Key == key {
                        h.buckets[index] = append(bucket[:i], bucket[i+1:]...)
                        h.size--
                        return entry.Value, true
                }
        }
        return "", false
}

func (h *HashTable) Contains(key string) bool {
        _, ok := h.Get(key)
        return ok
}

func (h *HashTable) Entries() []hashEntry {
        result := make([]hashEntry, 0, h.size)
        for _, bucket := range h.buckets {
                result = append(result, bucket...)
        }
        sort.Slice(result, func(i, j int) bool { return result[i].Key < result[j].Key })
        return result
}

func (h *HashTable) BucketCount() int { return len(h.buckets) }
func (h *HashTable) Collisions() int  { return h.collisions }

func (h *HashTable) LoadFactor() float64 {
        if len(h.buckets) == 0 {
                return 0
        }
        return float64(h.size) / float64(len(h.buckets))
}

func (h *HashTable) String() string {
        parts := make([]string, 0, h.size)
        for _, entry := range h.Entries() {
                parts = append(parts, entry.Key+"="+entry.Value)
        }
        return "{" + strings.Join(parts, ", ") + "}"
}

func (h *HashTable) bucketIndex(key string) int {
        return int(fnv1a(key) % uint64(len(h.buckets)))
}

func (h *HashTable) rehash(newCapacity int) {
        if newCapacity < 2 {
                newCapacity = 2
        }
        old := h.buckets
        h.buckets = make([][]hashEntry, newCapacity)
        h.size = 0
        h.collisions = 0
        for _, bucket := range old {
                for _, entry := range bucket {
                        h.Put(entry.Key, entry.Value)
                }
        }
}

func fnv1a(value string) uint64 {
        hash := uint64(1469598103934665603)
        for i := 0; i < len(value); i++ {
                hash ^= uint64(value[i])
                hash *= 1099511628211
        }
        return hash
}

func (h *HashTable) MarshalJSON() ([]byte, error) {
        return json.Marshal(h.Entries())
}

func (h *HashTable) UnmarshalJSON(data []byte) error {
        var entries []hashEntry
        if err := json.Unmarshal(data, &entries); err != nil {
                return fmt.Errorf("hash: %w", err)
        }
        h.Clear()
        for _, entry := range entries {
                h.Put(entry.Key, entry.Value)
        }
        return nil
}

// Queue — очередь FIFO. Используется кольцевой буфер на срезе:
// front указывает на первый элемент, данные не сдвигаются при pop.
type Queue struct {
        name  string
        data  []string
        front int
}

func NewQueue(name string) *Queue {
        return &Queue{name: name, data: make([]string, 0)}
}

func (q *Queue) Type() string { return TypeQueue }
func (q *Queue) Name() string { return q.name }
func (q *Queue) Len() int     { return len(q.data) - q.front }

func (q *Queue) Clear() {
        q.data = q.data[:0]
        q.front = 0
}

func (q *Queue) Push(value string) { q.data = append(q.data, value) }

func (q *Queue) Pop() (string, bool) {
        if q.front >= len(q.data) {
                return "", false
        }
        value := q.data[q.front]
        q.front++
        if q.front > 0 && q.front*2 >= len(q.data) {
                q.data = append([]string(nil), q.data[q.front:]...)
                q.front = 0
        }
        return value, true
}

func (q *Queue) Front() (string, bool) {
        if q.front >= len(q.data) {
                return "", false
        }
        return q.data[q.front], true
}

func (q *Queue) Values() []string {
        result := make([]string, 0, q.Len())
        result = append(result, q.data[q.front:]...)
        return result
}

func (q *Queue) String() string {
        values := q.Values()
        if len(values) == 0 {
                return "начало -> <- конец"
        }
        return "начало -> " + strings.Join(values, " ") + " <- конец"
}

func (q *Queue) MarshalJSON() ([]byte, error) {
        return json.Marshal(q.Values())
}

func (q *Queue) UnmarshalJSON(data []byte) error {
        var values []string
        if err := json.Unmarshal(data, &values); err != nil {
                return fmt.Errorf("queue: %w", err)
        }
        q.Clear()
        q.data = append(q.data, values...)
        return nil
}

        Type    string          `json:"type"`
        Name    string          `json:"name"`
        Payload json.RawMessage `json:"payload"`
}

type envelope struct {
        Magic   string   `json:"magic"`
        Version int      `json:"version"`
        Items   []record `json:"items"`
}

// NewByType создаёт контейнер указанного типа с заданным именем.
func NewByType(typeName, name string) (Container, error) {
        switch typeName {
        case TypeArray:
                return NewArray(name), nil
        case TypeList:
                return NewSinglyLinkedList(name), nil
        case TypeDList:
                return NewDoublyLinkedList(name), nil
        case TypeStack:
                return NewStack(name), nil
        case TypeQueue:
                return NewQueue(name), nil
        case TypeHash:
                return NewHashTable(name, 16), nil
        case TypeTree:
                return NewFullBinaryTree(name), nil
        default:
                return nil, fmt.Errorf("неизвестный тип структуры %q", typeName)
        }
}

// SaveText записывает контейнеры в текстовый JSON-файл. Имена сортируются,
// чтобы одинаковый набор данных давал одинаковый файл.
func SaveText(path string, items []Container) error {
        ordered := append([]Container(nil), items...)
        sort.Slice(ordered, func(i, j int) bool { return ordered[i].Name() < ordered[j].Name() })

        records := make([]record, 0, len(ordered))
        for _, item := range ordered {
                if item.Name() == "" {
                        return errors.New("имя структуры не может быть пустым")
                }
                payload, err := json.Marshal(item)
                if err != nil {
                        return fmt.Errorf("сериализация %q: %w", item.Name(), err)
                }
                records = append(records, record{Type: item.Type(), Name: item.Name(), Payload: payload})
        }
        data, err := json.MarshalIndent(envelope{
                Magic: formatMagic, Version: formatVersion, Items: records,
        }, "", "  ")
        if err != nil {
                return err
        }
        return os.WriteFile(path, append(data, '\n'), 0o644)
}

// LoadText читает JSON-файл и восстанавливает контейнеры по полю type.
func LoadText(path string) ([]Container, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                return nil, fmt.Errorf("не удалось прочитать файл: %w", err)
        }
        var input envelope
        if err := json.Unmarshal(data, &input); err != nil {
                return nil, fmt.Errorf("неверный формат JSON: %w", err)
        }
        if input.Magic != formatMagic {
                return nil, fmt.Errorf("неверная сигнатура формата %q", input.Magic)
        }
        if input.Version != formatVersion {
                return nil, fmt.Errorf("неподдерживаемая версия формата %d", input.Version)
        }

        result := make([]Container, 0, len(input.Items))
        names := make(map[string]bool, len(input.Items))
        for _, item := range input.Items {
                if item.Name == "" {
                        return nil, errors.New("пустое имя структуры в файле")
                }
                if names[item.Name] {
                        return nil, fmt.Errorf("повторяющееся имя %q", item.Name)
                }
                names[item.Name] = true
                container, err := NewByType(item.Type, item.Name)
                if err != nil {
                        return nil, err
                }
                if err := json.Unmarshal(item.Payload, container); err != nil {
                        return nil, fmt.Errorf("повреждённые данные %q: %w", item.Name, err)
                }
                result = append(result, container)
        }
        return result, nil
}

package containers

import (
        "bytes"
        "encoding/binary"
        "errors"
        "fmt"
        "os"
        "sort"
)

const (
        binaryMagic        = "DS3BIN"
        binaryVersion      = 1
        maxBinaryItems     = 1000000
        maxBinaryString    = 16 * 1024 * 1024
        maxBinaryPayload   = 64 * 1024 * 1024
        minBinaryTreeValue = -1 << 31
        maxBinaryTreeValue = 1<<31 - 1
)

// SaveBinary записывает контейнеры в бинарный файл с сигнатурой DS3BIN.
// Имена сортируются, чтобы одинаковый набор данных давал одинаковый файл.
func SaveBinary(path string, items []Container) error {
        if len(items) > maxBinaryItems {
                return fmt.Errorf("слишком много структур: %d", len(items))
        }
        for _, item := range items {
                if item == nil {
                        return errors.New("структура не может быть nil")
                }
        }
        ordered := append([]Container(nil), items...)
        sort.Slice(ordered, func(i, j int) bool { return ordered[i].Name() < ordered[j].Name() })

        var out bytes.Buffer
        out.WriteString(binaryMagic)
        writeBinaryUint32(&out, binaryVersion)
        writeBinaryUint32(&out, uint32(len(ordered)))

        names := make(map[string]bool, len(ordered))
        for _, item := range ordered {
                if item.Name() == "" {
                        return errors.New("имя структуры не может быть пустым")
                }
                if names[item.Name()] {
                        return fmt.Errorf("повторяющееся имя %q", item.Name())
                }
                names[item.Name()] = true

                payload, err := encodeBinaryPayload(item)
                if err != nil {
                        return fmt.Errorf("сериализация %q: %w", item.Name(), err)
                }
                if len(payload) > maxBinaryPayload {
                        return fmt.Errorf("слишком большой payload для %q", item.Name())
                }
                if err := writeBinaryString(&out, item.Type()); err != nil {
                        return fmt.Errorf("сериализация %q: %w", item.Name(), err)
                }
                if err := writeBinaryString(&out, item.Name()); err != nil {
                        return fmt.Errorf("сериализация %q: %w", item.Name(), err)
                }
                writeBinaryUint32(&out, uint32(len(payload)))
                out.Write(payload)
        }
        return os.WriteFile(path, out.Bytes(), 0o644)
}

// LoadBinary читает бинарный файл DS3BIN и восстанавливает контейнеры
// по полю type. При ошибке возвращается error без частичного результата.
func LoadBinary(path string) ([]Container, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                return nil, fmt.Errorf("не удалось прочитать файл: %w", err)
        }
        reader := binaryReader{data: data}
        if magic := reader.readBytes(len(binaryMagic)); string(magic) != binaryMagic {
                return nil, fmt.Errorf("неверная сигнатура формата")
        }
        if version := reader.readUint32(); version != binaryVersion {
                return nil, fmt.Errorf("неподдерживаемая версия формата %d", version)
        }
        count := reader.readUint32()
        if reader.err != nil {
                return nil, fmt.Errorf("неверный формат бинарного файла: %w", reader.err)
        }
        if count > maxBinaryItems {
                return nil, fmt.Errorf("слишком много структур в файле: %d", count)
        }

        result := make([]Container, 0, count)
        names := make(map[string]bool, count)
        for i := uint32(0); i < count; i++ {
                typeName := reader.readString()
                name := reader.readString()
                payloadLength := reader.readUint32()
                if reader.err != nil {
                        return nil, fmt.Errorf("неверный формат бинарного файла: %w", reader.err)
                }
                if payloadLength > maxBinaryPayload {
                        return nil, fmt.Errorf("слишком большой payload у записи %d", i)
                }
                payload := reader.readBytes(int(payloadLength))
                if reader.err != nil {
                        return nil, fmt.Errorf("неверный формат бинарного файла: %w", reader.err)
                }
                if name == "" {
                        return nil, errors.New("пустое имя структуры в файле")
                }
                if names[name] {
                        return nil, fmt.Errorf("повторяющееся имя %q", name)
                }
                names[name] = true

                item, err := NewByType(typeName, name)
                if err != nil {
                        return nil, err
                }
                if err := decodeBinaryPayload(item, payload); err != nil {
                        return nil, fmt.Errorf("повреждённые данные %q: %w", name, err)
                }
                result = append(result, item)
        }
        if reader.err != nil {
                return nil, fmt.Errorf("неверный формат бинарного файла: %w", reader.err)
        }
        if !reader.eof() {
                return nil, errors.New("лишние данные после последней структуры")
        }
        return result, nil
}

func encodeBinaryPayload(item Container) ([]byte, error) {
        switch item.Type() {
        case TypeArray:
                value, ok := item.(*Array)
                if !ok {
                        return nil, errors.New("неверный тип контейнера array")
                }
                return encodeBinarySequence("ARR1", value.data)
        case TypeList:
                value, ok := item.(*SinglyLinkedList)
                if !ok {
                        return nil, errors.New("неверный тип контейнера list")
                }
                return encodeBinarySequence("LST1", value.Values())
        case TypeDList:
                value, ok := item.(*DoublyLinkedList)
                if !ok {
                        return nil, errors.New("неверный тип контейнера dlist")
                }
                return encodeBinarySequence("DLST", value.Values())
        case TypeStack:
                value, ok := item.(*Stack)
                if !ok {
                        return nil, errors.New("неверный тип контейнера stack")
                }
                return encodeBinarySequence("STK1", value.data)
        case TypeQueue:
                value, ok := item.(*Queue)
                if !ok {
                        return nil, errors.New("неверный тип контейнера queue")
                }
                return encodeBinarySequence("QUE1", value.Values())
        case TypeHash:
                value, ok := item.(*HashTable)
                if !ok {
                        return nil, errors.New("неверный тип контейнера hash")
                }
                return encodeBinaryHash(value)
        case TypeTree:
                value, ok := item.(*FullBinaryTree)
                if !ok {
                        return nil, errors.New("неверный тип контейнера tree")
                }
                return encodeBinaryTree(value)
        default:
                return nil, fmt.Errorf("неизвестный тип структуры %q", item.Type())
        }
}

func decodeBinaryPayload(item Container, payload []byte) error {
        switch item.Type() {
        case TypeArray:
                value, ok := item.(*Array)
                if !ok {
                        return errors.New("неверный тип контейнера array")
                }
                values, err := decodeBinarySequence(payload, "ARR1")
                if err != nil {
                        return err
                }
                value.data = values
                return nil
        case TypeList:
                value, ok := item.(*SinglyLinkedList)
                if !ok {
                        return errors.New("неверный тип контейнера list")
                }
                values, err := decodeBinarySequence(payload, "LST1")
                if err != nil {
                        return err
                }
                value.Clear()
                for _, current := range values {
                        value.PushBack(current)
                }
                return nil
        case TypeDList:
                value, ok := item.(*DoublyLinkedList)
                if !ok {
                        return errors.New("неверный тип контейнера dlist")
                }
                values, err := decodeBinarySequence(payload, "DLST")
                if err != nil {
                        return err
                }
                value.Clear()
                for _, current := range values {
                        value.PushBack(current)
                }
                return nil
        case TypeStack:
                value, ok := item.(*Stack)
                if !ok {
                        return errors.New("неверный тип контейнера stack")
                }
                values, err := decodeBinarySequence(payload, "STK1")
                if err != nil {
                        return err
                }
                value.data = values
                return nil
        case TypeQueue:
                value, ok := item.(*Queue)
                if !ok {
                        return errors.New("неверный тип контейнера queue")
                }
                values, err := decodeBinarySequence(payload, "QUE1")
                if err != nil {
                        return err
                }
                value.Clear()
                value.data = append(value.data, values...)
                return nil
        case TypeHash:
                value, ok := item.(*HashTable)
                if !ok {
                        return errors.New("неверный тип контейнера hash")
                }
                return decodeBinaryHash(value, payload)
        case TypeTree:
                value, ok := item.(*FullBinaryTree)
                if !ok {
                        return errors.New("неверный тип контейнера tree")
                }
                return decodeBinaryTree(value, payload)
        default:
                return fmt.Errorf("неизвестный тип структуры %q", item.Type())
        }
}

func encodeBinarySequence(tag string, values []string) ([]byte, error) {
        if len(values) > maxBinaryItems {
                return nil, fmt.Errorf("слишком много элементов: %d", len(values))
        }
        var out bytes.Buffer
        out.WriteString(tag)
        writeBinaryUint32(&out, uint32(len(values)))
        for _, value := range values {
                if err := writeBinaryString(&out, value); err != nil {
                        return nil, err
                }
        }
        return out.Bytes(), nil
}

func decodeBinarySequence(payload []byte, tag string) ([]string, error) {
        reader := binaryReader{data: payload}
        if !reader.readTag(tag) {
                return nil, fmt.Errorf("неверный тег формата, ожидался %s", tag)
        }
        count := reader.readUint32()
        if reader.err != nil {
                return nil, reader.err
        }
        if count > maxBinaryItems {
                return nil, fmt.Errorf("слишком много элементов: %d", count)
        }
        values := make([]string, 0, count)
        for i := uint32(0); i < count; i++ {
                values = append(values, reader.readString())
                if reader.err != nil {
                        return nil, reader.err
                }
        }
        if !reader.eof() {
                return nil, errors.New("лишние данные после последовательности")
        }
        return values, nil
}

func encodeBinaryHash(table *HashTable) ([]byte, error) {
        entries := table.Entries()
        if len(entries) > maxBinaryItems {
                return nil, fmt.Errorf("слишком много пар: %d", len(entries))
        }
        var out bytes.Buffer
        out.WriteString("HSH1")
        writeBinaryUint32(&out, uint32(len(entries)))
        for _, entry := range entries {
                if err := writeBinaryString(&out, entry.Key); err != nil {
                        return nil, err
                }
                if err := writeBinaryString(&out, entry.Value); err != nil {
                        return nil, err
                }
        }
        return out.Bytes(), nil
}

func decodeBinaryHash(table *HashTable, payload []byte) error {
        reader := binaryReader{data: payload}
        if !reader.readTag("HSH1") {
                return errors.New("неверный тег формата, ожидался HSH1")
        }
        count := reader.readUint32()
        if reader.err != nil {
                return reader.err
        }
        if count > maxBinaryItems {
                return fmt.Errorf("слишком много пар: %d", count)
        }
        table.Clear()
        for i := uint32(0); i < count; i++ {
                key := reader.readString()
                value := reader.readString()
                if reader.err != nil {
                        return reader.err
                }
                table.Put(key, value)
        }
        if !reader.eof() {
                return errors.New("лишние данные после хеш-таблицы")
        }
        return nil
}

func encodeBinaryTree(tree *FullBinaryTree) ([]byte, error) {
        tokens := tree.PreorderWithNulls()
        if len(tokens) > maxBinaryItems {
                return nil, fmt.Errorf("слишком много токенов дерева: %d", len(tokens))
        }
        var out bytes.Buffer
        out.WriteString("TRE1")
        writeBinaryUint32(&out, uint32(len(tokens)))
        for _, token := range tokens {
                if token == nil {
                        out.WriteByte(0)
                        continue
                }
                if int64(*token) < minBinaryTreeValue || int64(*token) > maxBinaryTreeValue {
                        return nil, fmt.Errorf("значение дерева вне диапазона int32: %d", *token)
                }
                out.WriteByte(1)
                writeBinaryUint32(&out, uint32(int32(*token)))
        }
        return out.Bytes(), nil
}

func decodeBinaryTree(tree *FullBinaryTree, payload []byte) error {
        reader := binaryReader{data: payload}
        if !reader.readTag("TRE1") {
                return errors.New("неверный тег формата, ожидался TRE1")
        }
        count := reader.readUint32()
        if reader.err != nil {
                return reader.err
        }
        if count > maxBinaryItems {
                return fmt.Errorf("слишком много токенов дерева: %d", count)
        }
        tokens := make([]*int, 0, count)
        for i := uint32(0); i < count; i++ {
                marker := reader.readByte()
                if reader.err != nil {
                        return reader.err
                }
                switch marker {
                case 0:
                        tokens = append(tokens, nil)
                case 1:
                        raw := reader.readUint32()
                        if reader.err != nil {
                                return reader.err
                        }
                        value := int(int32(raw))
                        tokens = append(tokens, &value)
                default:
                        return fmt.Errorf("неверный маркер дерева: %d", marker)
                }
        }
        if !reader.eof() {
                return errors.New("лишние данные после дерева")
        }
        if !tree.LoadPreorderWithNulls(tokens) {
                return errors.New("некорректный preorder-формат дерева")
        }
        return nil
}

func writeBinaryUint32(out *bytes.Buffer, value uint32) {
        var data [4]byte
        binary.LittleEndian.PutUint32(data[:], value)
        out.Write(data[:])
}

func writeBinaryString(out *bytes.Buffer, value string) error {
        if len(value) > maxBinaryString {
                return fmt.Errorf("слишком длинная строка: %d", len(value))
        }
        writeBinaryUint32(out, uint32(len(value)))
        out.WriteString(value)
        return nil
}

type binaryReader struct {
        data   []byte
        offset int
        err    error
}

func (r *binaryReader) readBytes(count int) []byte {
        if r.err != nil {
                return nil
        }
        if count < 0 || r.offset+count > len(r.data) {
                r.err = errors.New("неожиданный конец файла")
                return nil
        }
        value := r.data[r.offset : r.offset+count]
        r.offset += count
        return value
}

func (r *binaryReader) readByte() byte {
        value := r.readBytes(1)
        if r.err != nil {
                return 0
        }
        return value[0]
}

func (r *binaryReader) readUint32() uint32 {
        value := r.readBytes(4)
        if r.err != nil {
                return 0
        }
        return binary.LittleEndian.Uint32(value)
}

func (r *binaryReader) readString() string {
        length := r.readUint32()
        if r.err != nil {
                return ""
        }
        if length > maxBinaryString {
                r.err = fmt.Errorf("слишком длинная строка: %d", length)
                return ""
        }
        return string(r.readBytes(int(length)))
}

func (r *binaryReader) readTag(tag string) bool {
        value := r.readBytes(len(tag))
        return r.err == nil && string(value) == tag
}

func (r *binaryReader) eof() bool {
        return r.err == nil && r.offset == len(r.data)
}

package containers

import (
        "encoding/json"
        "fmt"
        "strings"
)

type singlyNode struct {
        value string
        next  *singlyNode
}

// SinglyLinkedList — односвязный список с указателями head и tail.
type SinglyLinkedList struct {
        name string
        head *singlyNode
        tail *singlyNode
        size int
}

func NewSinglyLinkedList(name string) *SinglyLinkedList {
        return &SinglyLinkedList{name: name}
}

func (l *SinglyLinkedList) Type() string { return TypeList }
func (l *SinglyLinkedList) Name() string { return l.name }
func (l *SinglyLinkedList) Len() int     { return l.size }

func (l *SinglyLinkedList) Clear() {
        l.head = nil
        l.tail = nil
        l.size = 0
}

func (l *SinglyLinkedList) PushFront(value string) {
        node := &singlyNode{value: value, next: l.head}
        l.head = node
        if l.tail == nil {
                l.tail = node
        }
        l.size++
}

func (l *SinglyLinkedList) PushBack(value string) {
        node := &singlyNode{value: value}
        if l.tail == nil {
                l.head = node
        } else {
                l.tail.next = node
        }
        l.tail = node
        l.size++
}

func (l *SinglyLinkedList) InsertBefore(target, value string) bool {
        if l.head == nil {
                return false
        }
        if l.head.value == target {
                l.PushFront(value)
                return true
        }
        previous := l.head
        for previous.next != nil && previous.next.value != target {
                previous = previous.next
        }
        if previous.next == nil {
                return false
        }
        previous.next = &singlyNode{value: value, next: previous.next}
        l.size++
        return true
}

func (l *SinglyLinkedList) InsertAfter(target, value string) bool {
        node := l.findNode(target)
        if node == nil {
                return false
        }
        node.next = &singlyNode{value: value, next: node.next}
        if node == l.tail {
                l.tail = node.next
        }
        l.size++
        return true
}

func (l *SinglyLinkedList) PopFront() (string, bool) {
        if l.head == nil {
                return "", false
        }
        node := l.head
        l.head = node.next
        if l.head == nil {
                l.tail = nil
        }
        l.size--
        return node.value, true
}

func (l *SinglyLinkedList) PopBack() (string, bool) {
        if l.head == nil {
                return "", false
        }
        if l.head == l.tail {
                value := l.head.value
                l.Clear()
                return value, true
        }
        previous := l.head
        for previous.next != l.tail {
                previous = previous.next
        }
        value := l.tail.value
        previous.next = nil
        l.tail = previous
        l.size--
        return value, true
}

func (l *SinglyLinkedList) DeleteBefore(target string) (string, bool) {
        if l.head == nil || l.head.value == target {
                return "", false
        }
        previous := l.head
        for previous.next != nil && previous.next.value != target {
                previous = previous.next
        }
        if previous.next == nil {
                return "", false
        }
        if previous == l.head {
                return l.PopFront()
        }
        beforePrevious := l.head
        for beforePrevious.next != previous {
                beforePrevious = beforePrevious.next
        }
        value := previous.value
        beforePrevious.next = previous.next
        l.size--
        return value, true
}

func (l *SinglyLinkedList) DeleteAfter(target string) (string, bool) {
        node := l.findNode(target)
        if node == nil || node.next == nil {
                return "", false
        }
        removed := node.next
        node.next = removed.next
        if removed == l.tail {
                l.tail = node
        }
        l.size--
        return removed.value, true
}

func (l *SinglyLinkedList) DeleteValue(value string) (string, bool) {
        if l.head == nil {
                return "", false
        }
        if l.head.value == value {
                return l.PopFront()
        }
        previous := l.head
        for previous.next != nil && previous.next.value != value {
                previous = previous.next
        }
        if previous.next == nil {
                return "", false
        }
        removed := previous.next
        previous.next = removed.next
        if removed == l.tail {
                l.tail = previous
        }
        l.size--
        return removed.value, true
}

func (l *SinglyLinkedList) Find(value string) int {
        index := 0
        for node := l.head; node != nil; node = node.next {
                if node.value == value {
                        return index
                }
                index++
        }
        return -1
}

func (l *SinglyLinkedList) GetAt(index int) (string, bool) {
        node := l.nodeAt(index)
        if node == nil {
                return "", false
        }
        return node.value, true
}

func (l *SinglyLinkedList) SetAt(index int, value string) bool {
        node := l.nodeAt(index)
        if node == nil {
                return false
        }
        node.value = value
        return true
}

func (l *SinglyLinkedList) Values() []string {
        result := make([]string, 0, l.size)
        for node := l.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

func (l *SinglyLinkedList) String() string {
        if l.size == 0 {
                return "[]"
        }
        return "[" + strings.Join(l.Values(), " -> ") + "]"
}

func (l *SinglyLinkedList) nodeAt(index int) *singlyNode {
        if index < 0 || index >= l.size {
                return nil
        }
        node := l.head
        for i := 0; i < index; i++ {
                node = node.next
        }
        return node
}

func (l *SinglyLinkedList) findNode(value string) *singlyNode {
        for node := l.head; node != nil; node = node.next {
                if node.value == value {
                        return node
                }
        }
        return nil
}

func (l *SinglyLinkedList) MarshalJSON() ([]byte, error) {
        return json.Marshal(l.Values())
}

func (l *SinglyLinkedList) UnmarshalJSON(data []byte) error {
        var values []string
        if err := json.Unmarshal(data, &values); err != nil {
                return fmt.Errorf("list: %w", err)
        }
        l.Clear()
        for _, value := range values {
                l.PushBack(value)
        }
        return nil
}

package containers

import (
        "encoding/json"
        "fmt"
        "strings"
)

// Stack — стек LIFO на срезе Go. Вершина хранится в конце среза.
type Stack struct {
        name string
        data []string
}

func NewStack(name string) *Stack {
        return &Stack{name: name, data: make([]string, 0)}
}

func (s *Stack) Type() string { return TypeStack }
func (s *Stack) Name() string { return s.name }
func (s *Stack) Len() int     { return len(s.data) }
func (s *Stack) Clear()       { s.data = s.data[:0] }

func (s *Stack) Push(value string) { s.data = append(s.data, value) }

func (s *Stack) Pop() (string, bool) {
        if len(s.data) == 0 {
                return "", false
        }
        value := s.data[len(s.data)-1]
        s.data = s.data[:len(s.data)-1]
        return value, true
}

func (s *Stack) Peek() (string, bool) {
        if len(s.data) == 0 {
                return "", false
        }
        return s.data[len(s.data)-1], true
}

func (s *Stack) Values() []string {
        result := make([]string, len(s.data))
        for i, value := range s.data {
                result[len(s.data)-1-i] = value
        }
        return result
}

func (s *Stack) String() string {
        if len(s.data) == 0 {
                return "[]"
        }
        values := s.Values()
        return "[" + strings.Join(values, ", ") + "]"
}

func (s *Stack) MarshalJSON() ([]byte, error) {
        if s.data == nil {
                return []byte("[]"), nil
        }
        return json.Marshal(s.data)
}

func (s *Stack) UnmarshalJSON(data []byte) error {
        var values []string
        if err := json.Unmarshal(data, &values); err != nil {
                return fmt.Errorf("stack: %w", err)
        }
        s.data = values
        if s.data == nil {
                s.data = make([]string, 0)
        }
        return nil
}
